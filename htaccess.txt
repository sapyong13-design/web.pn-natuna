## Canonical Joomla rewrite rules for PN Natuna.
## Copy to .htaccess on deployment; keep both files synchronized.
Options +FollowSymlinks
Options -Indexes

<IfModule mod_autoindex.c>
	IndexIgnore *
</IfModule>
<IfModule mod_setenvif.c>
	SetEnvIf Host ^new\.pn-natuna\.go\.id$ STAGING
</IfModule>
<IfModule mod_headers.c>
	Header always set X-Content-Type-Options "nosniff"
	Header always set Referrer-Policy "strict-origin-when-cross-origin"
	Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"
	Header always unset X-Powered-By
	## One week without includeSubDomains keeps separately managed legacy apps safe.
	Header always set Strict-Transport-Security "max-age=604800" env=HTTPS
	Header always set X-Robots-Tag "noindex, nofollow, noarchive" env=STAGING
	Header always set Content-Security-Policy "default-src 'self'; base-uri 'self'; object-src 'none'; frame-ancestors 'self'; form-action 'self' https://www.google.com; script-src 'self' 'unsafe-inline' https://challenges.cloudflare.com https://www.instagram.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; img-src 'self' data: https:; font-src 'self' data: https://fonts.gstatic.com; frame-src 'self' https://challenges.cloudflare.com https://www.google.com https://www.instagram.com https://www.youtube-nocookie.com; connect-src 'self' https://www.instagram.com; media-src 'self'; worker-src 'self' blob:; upgrade-insecure-requests"
</IfModule>
<FilesMatch "\.svg$">
	<IfModule mod_headers.c>
		Header always set Content-Security-Policy "script-src 'none'"
	</IfModule>
</FilesMatch>

<IfModule mod_filter.c>
	<IfModule mod_deflate.c>
		AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css
		AddOutputFilterByType DEFLATE application/javascript application/x-javascript text/javascript
		AddOutputFilterByType DEFLATE application/json application/xml application/rss+xml
		AddOutputFilterByType DEFLATE image/svg+xml application/vnd.ms-fontobject
	</IfModule>
	<IfModule mod_brotli.c>
		AddOutputFilterByType BROTLI_COMPRESS text/html text/plain text/xml text/css
		AddOutputFilterByType BROTLI_COMPRESS application/javascript text/javascript
		AddOutputFilterByType BROTLI_COMPRESS application/json application/xml image/svg+xml
	</IfModule>
</IfModule>
<IfModule mod_setenvif.c>
	SetEnvIfNoCase Request_URI \.(?:gif|jpe?g|png|webp|avif|ico|zip|gz|woff2?)$ no-gzip dont-vary
</IfModule>
<IfModule mod_expires.c>
	ExpiresActive On
	ExpiresByType text/css "access plus 30 days"
	ExpiresByType text/javascript "access plus 30 days"
	ExpiresByType application/javascript "access plus 30 days"
	ExpiresByType font/woff2 "access plus 1 year"
	ExpiresByType font/woff "access plus 1 year"
	ExpiresByType image/webp "access plus 1 year"
	ExpiresByType image/avif "access plus 1 year"
	ExpiresByType image/jpeg "access plus 1 year"
	ExpiresByType image/png "access plus 1 year"
	ExpiresByType image/svg+xml "access plus 1 year"
	ExpiresByType image/x-icon "access plus 1 year"
	ExpiresByType text/html "access plus 0 seconds"
</IfModule>

<IfModule mod_authz_core.c>
	<FilesMatch "(^\.|configuration\.php$|(?:^|\.)(?:env|ini|log|sql|sqlite|dump|bak|backup|old|orig|save|swp|pem|key|p12|pfx)(?:\.|$)|(?:composer|package)-lock\.json$|composer\.json$)">
		Require all denied
	</FilesMatch>
</IfModule>
<IfModule !mod_authz_core.c>
	<FilesMatch "(^\.|configuration\.php$|(?:^|\.)(?:env|ini|log|sql|sqlite|dump|bak|backup|old|orig|save|swp|pem|key|p12|pfx)(?:\.|$)|(?:composer|package)-lock\.json$|composer\.json$)">
		Order allow,deny
		Deny from all
	</FilesMatch>
</IfModule>

<IfModule mod_rewrite.c>
	RewriteEngine On
	RewriteCond %{HTTPS} !=on
	RewriteCond %{HTTP:X-Forwarded-Proto} !https [NC]
	RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L,NE]
	## Canonical image slugs for the three July/August 2026 news articles.
	RewriteRule ^images/PATCANIA\.jpg$ /images/berita/2026/alih-tugas-cania-kirana-1.webp [R=301,L,NE,NC]
	RewriteRule ^images/PATCANIA-(400|800|1200)\.webp$ /images/berita/2026/alih-tugas-cania-kirana-1-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3408\.jpg$ /images/berita/2026/alih-tugas-cania-kirana-2.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3408-(400|800|1200)\.webp$ /images/berita/2026/alih-tugas-cania-kirana-2-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/WhatsApp\x20Image\x202026-07-31\x20at\x2008\.17\.57\.jpeg$ /images/berita/2026/bola-voli-hut-81-ri-ma-1.webp [R=301,L,NE,NC]
	RewriteRule ^images/WhatsApp\x20Image\x202026-07-31\x20at\x2008\.17\.57-(400|800|1200)\.webp$ /images/berita/2026/bola-voli-hut-81-ri-ma-1-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3150\.jpg$ /images/berita/2026/bola-voli-hut-81-ri-ma-2.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3150-(400|800|1200)\.webp$ /images/berita/2026/bola-voli-hut-81-ri-ma-2-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3204\.jpg$ /images/berita/2026/bola-voli-hut-81-ri-ma-3.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3204-(400|800|1200)\.webp$ /images/berita/2026/bola-voli-hut-81-ri-ma-3-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3729\x201\.jpg$ /images/berita/2026/mobile-legends-hut-81-ri-ma-1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3729\x201-(400|800|1200)\.webp$ /images/berita/2026/mobile-legends-hut-81-ri-ma-1-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3701\.jpg$ /images/berita/2026/mobile-legends-hut-81-ri-ma-2.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3701-(400|800|1200)\.webp$ /images/berita/2026/mobile-legends-hut-81-ri-ma-2-$1.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3738\x201\.jpg$ /images/berita/2026/mobile-legends-hut-81-ri-ma-3.webp [R=301,L,NE,NC]
	RewriteRule ^images/IMG_3738\x201-(400|800|1200)\.webp$ /images/berita/2026/mobile-legends-hut-81-ri-ma-3-$1.webp [R=301,L,NE,NC]
	RewriteRule "(?:^|/)(?:\.git|\.svn|database|docs|tools)(?:/|$)" - [F,END,NC]
	RewriteRule ^api(?:/|$) - [F,END,NC]
	RewriteRule ^(?:images|files|cache|tmp)/.*\.(?:php[0-9]?|phtml|phar)$ - [F,END,NC]
	RewriteCond %{QUERY_STRING} base64_encode[^(]*\([^)]*\) [OR]
	RewriteCond %{QUERY_STRING} (<|%3C)([^s]*s)+cript.*(>|%3E) [NC,OR]
	RewriteCond %{QUERY_STRING} GLOBALS(=|\[|\%[0-9A-Z]{0,2}) [OR]
	RewriteCond %{QUERY_STRING} _REQUEST(=|\[|\%[0-9A-Z]{0,2})
	RewriteRule .* index.php [F]
	RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]
	RewriteCond %{REQUEST_URI} ^/api/
	RewriteCond %{REQUEST_URI} !^/api/index\.php
	RewriteCond %{REQUEST_FILENAME} !-f
	RewriteCond %{REQUEST_FILENAME} !-d
	RewriteRule .* api/index.php [L]
	RewriteCond %{REQUEST_URI} !^/index\.php
	RewriteCond %{REQUEST_FILENAME} !-f
	RewriteCond %{REQUEST_FILENAME} !-d
	RewriteRule .* index.php [L]
</IfModule>
<IfModule !mod_rewrite.c>
	<IfModule mod_alias.c>
		RedirectMatch 302 ^/$ /index.php/
	</IfModule>
</IfModule>
